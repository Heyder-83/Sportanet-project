// backend/test/authenticate.test.js
// Pruebas del middleware JWT. No requieren MySQL (findUserById se mockea).
// Ejecutar: npm test
const { test, before, after } = require('node:test');
const assert = require('node:assert/strict');
const path = require('node:path');
const { spawnSync } = require('node:child_process');

const SECRET = 'test-secret-only-for-unit-tests-0123456789abcdef';
process.env.JWT_SECRET = SECRET;

const express = require('express');
const jwt = require('jsonwebtoken');
const createAuthenticate = require('../middleware/authenticate');

let server, base;

// Usuario "vivo" por defecto para los tests que no especifiquen otra cosa.
const ACTIVE_USER = { is_active: 1, deleted_at: null };

before(async () => {
    const app = express();
    app.use(express.json());

    const defaultFindUser = async () => ACTIVE_USER;
    const defaultAuth = createAuthenticate({ findUserById: defaultFindUser });

    app.post('/protected', defaultAuth, (req, res) =>
        res.json({ auth: req.auth, claimed: req.body.user_id })
    );

    server = app.listen(0);
    await new Promise((r) => server.once('listening', r));
    base = `http://127.0.0.1:${server.address().port}`;
});

after(() => server.close());

const call = (token, rawHeader, body = {}) =>
    fetch(`${base}/protected`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            ...(rawHeader ? { Authorization: rawHeader } : token ? { Authorization: `Bearer ${token}` } : {})
        },
        body: JSON.stringify(body)
    });

// ---------- Tests existentes (firma y formato del token) ----------

test('token válido -> 200 e identidad tomada del token, no del body', async () => {
    const token = jwt.sign({ id: 7, email: 'a@b.co' }, SECRET, { expiresIn: '1h' });
    const res = await call(token, null, { user_id: 999 });
    assert.equal(res.status, 200);
    const json = await res.json();
    assert.deepEqual(json.auth, { userId: 7, email: 'a@b.co' });
    assert.equal(json.claimed, 999);
});

test('token ausente -> 401 TOKEN_MISSING', async () => {
    const res = await call();
    assert.equal(res.status, 401);
    assert.equal((await res.json()).code, 'TOKEN_MISSING');
});

test('header mal formado -> 401 TOKEN_MALFORMED', async () => {
    const res = await call(null, 'Token abc');
    assert.equal(res.status, 401);
    assert.equal((await res.json()).code, 'TOKEN_MALFORMED');
});

test('token inválido (basura) -> 401 TOKEN_INVALID', async () => {
    const res = await call('no.es.un-jwt');
    assert.equal(res.status, 401);
    assert.equal((await res.json()).code, 'TOKEN_INVALID');
});

test('token firmado con otro secreto -> 401 TOKEN_INVALID', async () => {
    const res = await call(jwt.sign({ id: 7, email: 'a@b.co' }, 'otro-secreto', { expiresIn: '1h' }));
    assert.equal(res.status, 401);
    assert.equal((await res.json()).code, 'TOKEN_INVALID');
});

test('token alg=none -> 401 TOKEN_INVALID', async () => {
    const b64 = (o) => Buffer.from(JSON.stringify(o)).toString('base64url');
    const forged = `${b64({ alg: 'none', typ: 'JWT' })}.${b64({ id: 1, email: 'x@y.z' })}.`;
    const res = await call(forged);
    assert.equal(res.status, 401);
    assert.equal((await res.json()).code, 'TOKEN_INVALID');
});

test('token expirado -> 401 TOKEN_EXPIRED', async () => {
    const token = jwt.sign({ id: 7, email: 'a@b.co' }, SECRET, { expiresIn: -60 });
    const res = await call(token);
    assert.equal(res.status, 401);
    assert.equal((await res.json()).code, 'TOKEN_EXPIRED');
});

test('token válido sin id utilizable -> 401 TOKEN_INVALID', async () => {
    const res = await call(jwt.sign({ email: 'a@b.co' }, SECRET, { expiresIn: '1h' }));
    assert.equal(res.status, 401);
    assert.equal((await res.json()).code, 'TOKEN_INVALID');
});

test('JWT_SECRET vacío -> config/jwt.js lanza error (sin depender del filesystem ni de .env)', () => {
    const r = spawnSync(process.execPath, ['-e', `require(${JSON.stringify(path.resolve(__dirname, '../config/jwt'))})`], {
        env: { ...process.env, JWT_SECRET: '' },
        encoding: 'utf8'
    });
    assert.notEqual(r.status, 0);
    assert.match(r.stderr, /JWT_SECRET no está definido/);
});

// ---------- Tests nuevos: estado de cuenta ----------

async function withFindUser(findUserById) {
    const app = express();
    app.use(express.json());
    const auth = createAuthenticate({ findUserById });
    app.post('/protected', auth, (req, res) => res.json({ auth: req.auth }));

    const srv = app.listen(0);
    await new Promise((r) => srv.once('listening', r));
    const url = `http://127.0.0.1:${srv.address().port}/protected`;
    return { srv, url };
}

const postWithToken = (url, token) =>
    fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
        body: '{}'
    });

test('usuario no existe en BD -> 401 TOKEN_INVALID', async () => {
    const { srv, url } = await withFindUser(async () => null);
    try {
        const token = jwt.sign({ id: 7, email: 'a@b.co' }, SECRET, { expiresIn: '1h' });
        const res = await postWithToken(url, token);
        assert.equal(res.status, 401);
        assert.equal((await res.json()).code, 'TOKEN_INVALID');
    } finally {
        srv.close();
    }
});

test('usuario con is_active=0 -> 401 ACCOUNT_DISABLED', async () => {
    const { srv, url } = await withFindUser(async () => ({ is_active: 0, deleted_at: null }));
    try {
        const token = jwt.sign({ id: 7, email: 'a@b.co' }, SECRET, { expiresIn: '1h' });
        const res = await postWithToken(url, token);
        assert.equal(res.status, 401);
        assert.equal((await res.json()).code, 'ACCOUNT_DISABLED');
    } finally {
        srv.close();
    }
});

test('usuario con deleted_at seteado -> 401 ACCOUNT_DELETED', async () => {
    const { srv, url } = await withFindUser(async () => ({ is_active: 1, deleted_at: new Date() }));
    try {
        const token = jwt.sign({ id: 7, email: 'a@b.co' }, SECRET, { expiresIn: '1h' });
        const res = await postWithToken(url, token);
        assert.equal(res.status, 401);
        assert.equal((await res.json()).code, 'ACCOUNT_DELETED');
    } finally {
        srv.close();
    }
});

test('findUserById lanza error -> 500 DB_ERROR (sin filtrar detalles)', async () => {
    const { srv, url } = await withFindUser(async () => { throw new Error('conexión perdida'); });
    try {
        const token = jwt.sign({ id: 7, email: 'a@b.co' }, SECRET, { expiresIn: '1h' });
        const res = await postWithToken(url, token);
        assert.equal(res.status, 500);
        const json = await res.json();
        assert.equal(json.code, 'DB_ERROR');
        assert.ok(!json.error.includes('conexión'), 'no debe filtrar el mensaje interno');
    } finally {
        srv.close();
    }
});
